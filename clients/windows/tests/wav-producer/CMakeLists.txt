cmake_minimum_required(VERSION 3.24)
project(OpenWebFixtureWavContract LANGUAGES C)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Only an isolated hosted Windows x64 MSVC contract build is planned")
endif()
set(OPENWEB_PINNED_SDK "" CACHE PATH "Exact SHA-verified 5.5.23 win64 directory on hosted Windows only")
foreach(header mediastreamer2/msfactory.h mediastreamer2/msfilter.h mediastreamer2/msasync.h mediastreamer2/msqueue.h bctoolbox/vfs.h ortp/str_utils.h)
  if(NOT EXISTS "${OPENWEB_PINNED_SDK}/include/${header}")
    message(FATAL_ERROR "Required real SDK public header absent: ${header}")
  endif()
endforeach()
foreach(library mediastreamer2 bctoolbox ortp)
  if(NOT EXISTS "${OPENWEB_PINNED_SDK}/lib/${library}.lib")
    message(FATAL_ERROR "Required real SDK import library absent: ${library}")
  endif()
endforeach()
add_library(openwebwav SHARED component/msfilerec.c component/asyncrw.c component/plugin.c)
set_target_properties(openwebwav PROPERTIES OUTPUT_NAME libmsopenwebwav)
target_compile_features(openwebwav PRIVATE c_std_11)
target_compile_definitions(openwebwav PRIVATE OPENWEB_FIXTURE_PLUGIN_BUILD _CRT_SECURE_NO_WARNINGS)
target_include_directories(openwebwav PRIVATE "${OPENWEB_PINNED_SDK}/include" component)
target_compile_options(openwebwav PRIVATE /W4)
target_link_libraries(openwebwav PRIVATE "${OPENWEB_PINNED_SDK}/lib/mediastreamer2.lib" "${OPENWEB_PINNED_SDK}/lib/bctoolbox.lib" "${OPENWEB_PINNED_SDK}/lib/ortp.lib")
add_executable(wav_contract contract/wav_contract.c)
target_compile_features(wav_contract PRIVATE c_std_11)
target_include_directories(wav_contract PRIVATE "${OPENWEB_PINNED_SDK}/include" component)
target_compile_definitions(wav_contract PRIVATE _CRT_SECURE_NO_WARNINGS)
target_compile_options(wav_contract PRIVATE /W4 /WX)
target_link_libraries(wav_contract PRIVATE openwebwav "${OPENWEB_PINNED_SDK}/lib/mediastreamer2.lib" "${OPENWEB_PINNED_SDK}/lib/bctoolbox.lib" "${OPENWEB_PINNED_SDK}/lib/ortp.lib")
